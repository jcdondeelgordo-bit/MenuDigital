/**
 * Funciones de la hoja "Clientes", usadas por bienvenida.html Y por el checkout de
 * Domicilio/mesa de menu.html (registrarClienteDesdeMenu, registrarPunto). Columnas de
 * Clientes: Fecha Ingreso, Telefono, Nombre, Cumpleanos, Visitas, Direccion (sin
 * tildes/eñe, confirmado por el usuario 2026-07-21; columna Direccion agregada
 * 2026-07-22), Puntos (agregada 2026-08-15, ver módulo 6 -- fidelización).
 *
 * Requiere que la hoja Clientes tenga columnas con encabezado exacto "Direccion" y
 * "Puntos". Si no existen, el código simplemente no guarda ese dato puntual (no rompe
 * el resto del registro de cliente).
 */

function buscarCliente(e) {
  const telefono = e.parameter.telefono || '';
  const hoja = SpreadsheetApp.getActive().getSheetByName('Clientes');
  const datos = hoja.getDataRange().getValues();
  const encabezados = datos[0];
  const idxTelefono = encabezados.indexOf('Telefono');
  const idxNombre = encabezados.indexOf('Nombre');
  const idxVisitas = encabezados.indexOf('Visitas');
  const idxDireccion = encabezados.indexOf('Direccion');
  const idxPuntos = encabezados.indexOf('Puntos');

  for (let i = 1; i < datos.length; i++) {
    if (String(datos[i][idxTelefono]) === String(telefono)) {
      return ContentService.createTextOutput(JSON.stringify({
        ok: true,
        existe: true,
        cliente: {
          nombre: datos[i][idxNombre],
          visitas: Number(datos[i][idxVisitas]) || 1,
          direccion: idxDireccion > -1 ? (datos[i][idxDireccion] || '') : '',
          puntos: idxPuntos > -1 ? (Number(datos[i][idxPuntos]) || 0) : 0
        }
      })).setMimeType(ContentService.MimeType.JSON);
    }
  }
  return ContentService.createTextOutput(JSON.stringify({ok: true, existe: false}))
    .setMimeType(ContentService.MimeType.JSON);
}

/**
 * accion=actualizar_cliente&telefono=...&direccion=... (direccion es opcional)
 * Suma 1 a "Visitas" cada vez que un cliente ya registrado vuelve (bienvenida.html, o
 * un nuevo pedido a Domicilio desde menu.html). Si llega "direccion", también la
 * actualiza a la más reciente — las direcciones pueden cambiar de un pedido a otro.
 */
function actualizarCliente(e) {
  const telefono = e.parameter.telefono || '';
  const direccion = e.parameter.direccion || '';
  const hoja = SpreadsheetApp.getActive().getSheetByName('Clientes');
  const datos = hoja.getDataRange().getValues();
  const encabezados = datos[0];
  const idxTelefono = encabezados.indexOf('Telefono');
  const idxVisitas = encabezados.indexOf('Visitas');
  const idxDireccion = encabezados.indexOf('Direccion');

  for (let i = 1; i < datos.length; i++) {
    if (String(datos[i][idxTelefono]) === String(telefono)) {
      const visitasActuales = Number(datos[i][idxVisitas]) || 0;
      hoja.getRange(i + 1, idxVisitas + 1).setValue(visitasActuales + 1);
      if (direccion && idxDireccion > -1) {
        hoja.getRange(i + 1, idxDireccion + 1).setValue(direccion);
      }
      break;
    }
  }
  return ContentService.createTextOutput(JSON.stringify({ok: true}))
    .setMimeType(ContentService.MimeType.JSON);
}

/**
 * accion=registrar_cliente&telefono=...&nombre=...&cumpleanos=...&canal=...&direccion=...
 * Crea un cliente nuevo en "Clientes" la primera vez que alguien lo usa.
 */
function registrarCliente(e) {
  const telefono = e.parameter.telefono || '';
  const nombre = e.parameter.nombre || '';
  const cumpleanos = e.parameter.cumpleanos || '';
  const direccion = e.parameter.direccion || '';

  const hoja = SpreadsheetApp.getActive().getSheetByName('Clientes');
  const datos = hoja.getDataRange().getValues();
  const encabezados = datos[0];
  const idxFecha = encabezados.indexOf('Fecha Ingreso');
  const idxTelefono = encabezados.indexOf('Telefono');
  const idxNombre = encabezados.indexOf('Nombre');
  const idxCumpleanos = encabezados.indexOf('Cumpleanos');
  const idxVisitas = encabezados.indexOf('Visitas');
  const idxDireccion = encabezados.indexOf('Direccion');
  const idxPuntos = encabezados.indexOf('Puntos');

  for (let i = 1; i < datos.length; i++) {
    if (String(datos[i][idxTelefono]) === String(telefono)) {
      return ContentService.createTextOutput(JSON.stringify({ok: true, yaExistia: true}))
        .setMimeType(ContentService.MimeType.JSON);
    }
  }

  const fila = new Array(encabezados.length).fill('');
  fila[idxFecha] = new Date();
  fila[idxTelefono] = telefono;
  fila[idxNombre] = nombre;
  fila[idxCumpleanos] = cumpleanos;
  fila[idxVisitas] = 1;
  if (idxDireccion > -1) fila[idxDireccion] = direccion;
  if (idxPuntos > -1) fila[idxPuntos] = 0;
  hoja.appendRow(fila);

  return ContentService.createTextOutput(JSON.stringify({ok: true, yaExistia: false}))
    .setMimeType(ContentService.MimeType.JSON);
}

/**
 * accion=registrar_punto&telefono=...&total=...&nombre=... (nombre opcional)
 * Único punto de entrada que otorga puntos de fidelidad -- reemplaza el contador de
 * localStorage que vivía en menu.html (sumarPuntoSiAplica), que se reseteaba solo y no
 * distinguía clientes reales. Rechaza si el total no supera $50.000 o si no hay
 * teléfono (sin identidad de cliente no hay a quién sumarle el punto). Si el cliente
 * no existe todavía en "Clientes", lo crea con Puntos=1 (mismo criterio que
 * registrarCliente, pero arrancando ya con su primer punto en vez de 0). Si ya existe,
 * le suma 1 a lo que ya tenía -- nunca lo resetea a 1 solo porque llegó un punto nuevo
 * (a diferencia del contador viejo, que reiniciaba el ciclo cada 12).
 */
function registrarPunto(e) {
  const telefono = e.parameter.telefono || '';
  const total = Number(e.parameter.total || 0);
  const nombre = e.parameter.nombre || '';

  if (!telefono || total < 50000) {
    return ContentService.createTextOutput(JSON.stringify({ ok: false, error: 'No aplica punto (falta teléfono o el total no supera $50.000)' }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  const hoja = SpreadsheetApp.getActive().getSheetByName('Clientes');
  const datos = hoja.getDataRange().getValues();
  const encabezados = datos[0];
  const idxTelefono = encabezados.indexOf('Telefono');
  const idxPuntos = encabezados.indexOf('Puntos');

  if (idxPuntos === -1) {
    return ContentService.createTextOutput(JSON.stringify({ ok: false, error: 'La hoja Clientes todavía no tiene la columna Puntos' }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  for (let i = 1; i < datos.length; i++) {
    if (String(datos[i][idxTelefono]) === String(telefono)) {
      const puntosActuales = Number(datos[i][idxPuntos]) || 0;
      const nuevoPuntos = puntosActuales >= 12 ? 1 : puntosActuales + 1;
      hoja.getRange(i + 1, idxPuntos + 1).setValue(nuevoPuntos);
      return ContentService.createTextOutput(JSON.stringify({ ok: true, puntos: nuevoPuntos }))
        .setMimeType(ContentService.MimeType.JSON);
    }
  }

  // Cliente nuevo (nunca pasó por registrar_cliente) -- se crea aquí mismo con su
  // primer punto, para no perder el punto si el flujo que llamó a esto no hizo
  // primero un registrar_cliente por separado.
  const idxFecha = encabezados.indexOf('Fecha Ingreso');
  const idxNombre = encabezados.indexOf('Nombre');
  const idxVisitas = encabezados.indexOf('Visitas');
  const fila = new Array(encabezados.length).fill('');
  fila[idxFecha] = new Date();
  fila[idxTelefono] = telefono;
  if (idxNombre > -1) fila[idxNombre] = nombre;
  if (idxVisitas > -1) fila[idxVisitas] = 1;
  fila[idxPuntos] = 1;
  hoja.appendRow(fila);

  return ContentService.createTextOutput(JSON.stringify({ ok: true, puntos: 1 }))
    .setMimeType(ContentService.MimeType.JSON);
}
